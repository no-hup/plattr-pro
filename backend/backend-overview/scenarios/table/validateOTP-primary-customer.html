{

    "restaurantId": "rest001",

    "otp": "123456",
    "phoneNumber": "+911234567890",
    "name": "Aditi",
    "userLocation": { "latitude": 12.9717, "longitude": 77.5945 }
  }
}

{
  "status": "success",
  "message": "Session created",
  "data": {
    "sessionId": "sess_table009_primary",
    "tableState": "ACTIVE",
    "primaryCustomer": {
      "customerId": "cust_aditi",
      "phoneNumber": "+911234567890"
    },
    "customToken": "eyJhbGciOiJSUzI1NiIsInR5cCI6IkpXVCJ9..."

Notes

  • Creates session + updates restaurants/{restaurantId}/tables/{tableId} document.
